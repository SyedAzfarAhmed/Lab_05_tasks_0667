#include <stdio.h>
int main()
{
    int category, subtype, delayed;
    printf("Chatbot Menu\n");
    printf("1=Greeting\n");
    printf("2=Query\n");
    printf("3=Complaint\n");
    printf("4=Feedback\n");
    printf("Choose category: ");
    scanf("%d", &category);
    switch(category)
    {
        case 1:
            printf("\nGreeting\n");
            printf("1=Morning\n");
            printf("2=Evening\n");
            printf("Choose sub-type: ");
            scanf("%d", &subtype);
            switch(subtype)
            {
                case 1:
                    printf("Good morning! How can I help you?\n");
                    break;
                case 2:
                    printf("Good evening! How can I help you?\n");
                    break;
                default:
                    printf("Invalid selection.\n");
            }
            break;
        case 2:
            printf("\nQuery\n");
            printf("1=Product\n");
            printf("2=Billing\n");
            printf("3=Technical\n");
            printf("Choose sub-type: ");
            scanf("%d", &subtype);
            switch(subtype)
            {
                case 1:
                    printf("Please provide the product details you want to know about.\n");
                    break;
                case 2:
                    printf("Please provide your billing details.\n");
                    break;
                case 3:
                    printf("Please describe your technical problem.\n");
                    break;
                default:
                    printf("Invalid selection.\n");
            }
            break;
        case 3:
            printf("\nComplaint\n");
            printf("1=Delivery\n");
            printf("2=Quality\n");
            printf("Choose sub-type: ");
            scanf("%d", &subtype);
            switch(subtype)
            {
                case 1:
                    printf("Is your order delayed? (1=Yes, 0=No): ");
                    scanf("%d", &delayed);
                    if(delayed==1)
                        printf("We sincerely apologize for the delay in your delivery.\n");
                    else if(delayed==0)
                        printf("We apologize for any inconvenience with your delivery.\n");
                    else
                        printf("Invalid selection.\n");
                    break;
                case 2:
                    printf("We apologize for the quality issue. We will look into it.\n");
                    break;
                default:
                    printf("Invalid selection.\n");
            }
            break;
        case 4:
            printf("\nFeedback\n");
            printf("1=Positive\n");
            printf("2=Negative\n");
            printf("Choose sub-type: ");
            scanf("%d", &subtype);
            switch(subtype)
            {
                case 1:
                    printf("Thank you for your positive feedback!\n");
                    break;
                case 2:
                    printf("Thank you for your feedback. We will work to improve.\n");
                    break;
                default:
                    printf("Invalid selection.\n");
            }
            break;
        default:
            printf("Invalid selection.\n");
    }
}
